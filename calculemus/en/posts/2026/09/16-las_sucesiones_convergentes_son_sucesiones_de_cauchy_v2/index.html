    Challenge 19: Convergent sequences are Cauchy sequences

    José A. Alonso

    16-09-2026

    Also available in:

    Español

    Prove that convergent sequences are Cauchy sequences

    To do so, complete the following Lean 4 theory:

    import Mathlib.Data.Real.Basic
import Mathlib.Tactic

variable {u : ℕ → ℝ}

def SeqLim (u : ℕ → ℝ) (a : ℝ) : Prop :=
  ∀ ε > 0, ∃ k, ∀ n ≥ k, |u n - a| < ε

def SeqConv (u : ℕ → ℝ) :=
  ∃ a, SeqLim u a

def IsCauchy (u : ℕ → ℝ) :=
  ∀ ε > 0, ∃ k, ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε

example
  (h : SeqConv u)
  : IsCauchy u :=
by sorry

    1. Proof in natural language

    Let \(\varepsilon \in \mathbb{R}\) be such that \(\varepsilon > 0\). We have to prove that there exists a \(k \in \mathbb{N}\) such that \[ \forall p \geq k, \forall q \geq k, |u(p) - u(q)| < \varepsilon \tag{1} \]

    Since \(u\) is convergent, there exists an \(a \in \mathbb{R}\) such that the limit of \(u\) is \(a\). Hence, there exists a \(k \in \mathbb{N}\) such that \[ \forall n \geq k, |u(n) - a| < \dfrac{\varepsilon}{2} \tag{2} \]

    To prove that (1) holds for this \(k\), let \(p, q \in \mathbb{N}\) be such that \(p \geq k\) and \(q \geq k\). Then, by (2), we have \[ \begin{align} |u_p - a| &< \dfrac{\varepsilon}{2} \tag{3} \\ |u_q - a| &< \dfrac{\varepsilon}{2} \tag{4} \end{align} \] Therefore, \[ \begin{array}{llll} |u_p - u_q| &= |(u_p - a) + (a - u_q)| \\ &≤ |u_p - a| + |a - u_q| \\ &= |u_p - a| + |u_q - a| \\ &< \dfrac{\varepsilon}{2} + \dfrac{\varepsilon}{2} &&\text{[by (3) and (4)]} \\ &= \varepsilon \end{array} \]

    2. Proofs in Lean 4

    import Mathlib.Data.Real.Basic
import Mathlib.Tactic

variable {u : ℕ → ℝ}

def SeqLim (u : ℕ → ℝ) (a : ℝ) : Prop :=
  ∀ ε > 0, ∃ k, ∀ n ≥ k, |u n - a| < ε

def SeqConv (u : ℕ → ℝ) :=
  ∃ a, SeqLim u a

def IsCauchy (u : ℕ → ℝ) :=
  ∀ ε > 0, ∃ k, ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε

-- Proof 1
-- =======

example
  (h : SeqConv u)
  : IsCauchy u :=
by
  intros ε hε
  -- ε : ℝ
  -- hε : ε > 0
  -- ⊢ ∃ k, ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  obtain ⟨a, ha⟩ := h
  -- a : ℝ
  -- ha : SeqLim u a
  obtain ⟨k, hk⟩ := ha (ε/2) (by grind)
  -- k : ℕ
  -- hk : ∀ n ≥ k, |u n - a| < ε / 2
  use k
  -- ⊢ ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  intros p hp q hq
  -- p : ℕ
  -- hp : p ≥ k
  -- q : ℕ
  -- hq : q ≥ k
  -- ⊢ |u p - u q| < ε
  grind

-- Proof 2
-- =======

example
  (h : SeqConv u)
  : IsCauchy u :=
by
  intros ε hε
  -- ε : ℝ
  -- hε : ε > 0
  -- ⊢ ∃ k, ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  obtain ⟨a, ha⟩ := h
  -- a : ℝ
  -- ha : SeqLim u a
  obtain ⟨k, hk⟩ := ha (ε/2) (by grind)
  -- k : ℕ
  -- hk : ∀ n ≥ k, |u n - a| < ε / 2
  use k
  -- ⊢ ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  intros p hp q hq
  -- p : ℕ
  -- hp : p ≥ k
  -- q : ℕ
  -- hq : q ≥ k
  -- ⊢ |u p - u q| < ε
  calc |u p - u q|
       = |(u p - a) + (a - u q)| := by grind
     _ ≤ |u p - a|  + |a - u q|  := by grind
     _ = |u p - a|  + |u q - a|  := by grind
     _ < ε                       := by grind

-- Proof 3
-- =======

example
  (h : SeqConv u)
  : IsCauchy u :=
by
  intros ε hε
  -- ε : ℝ
  -- hε : ε > 0
  -- ⊢ ∃ k, ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  obtain ⟨a, ha⟩ := h
  -- a : ℝ
  -- ha : SeqLim u a
  obtain ⟨k, hk⟩ := ha (ε/2) (by positivity)
  -- k : ℕ
  -- hk : ∀ n ≥ k, |u n - a| < ε / 2
  use k
  -- ⊢ ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  intros p hp q hq
  -- p : ℕ
  -- hp : p ≥ k
  -- q : ℕ
  -- hq : q ≥ k
  -- ⊢ |u p - u q| < ε
  have h1 : |u p - a| < ε / 2 := hk p hp
  have h2 : |u q - a| < ε / 2 := hk q hq
  calc |u p - u q|
       = |(u p - a) + (a - u q)| := by ring_nf
     _ ≤ |u p - a| + |a - u q|   := abs_add_le _ _
     _ = |u p - a| + |u q - a|   := by simp [abs_sub_comm]
     _ < ε / 2 + ε / 2           := by gcongr
     _ = ε                       := by ring

-- Proof 4
-- =======

example
  (h : SeqConv u)
  : IsCauchy u :=
by
  intros ε hε
  -- ε : ℝ
  -- hε : ε > 0
  -- ⊢ ∃ k, ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  obtain ⟨a, ha⟩ := h
  -- a : ℝ
  -- ha : SeqLim u a
  obtain ⟨k, hk⟩ := ha (ε/2) (half_pos hε)
  -- k : ℕ
  -- hk : ∀ n ≥ k, |u n - a| < ε / 2
  use k
  -- ⊢ ∀ p ≥ k, ∀ q ≥ k, |u p - u q| < ε
  intros p hp q hq
  -- p : ℕ
  -- hp : p ≥ k
  -- q : ℕ
  -- hq : q ≥ k
  -- ⊢ |u p - u q| < ε
  calc |u p - u q|
       = |(u p - a) + (a - u q)| :=
           congrArg abs (sub_add_sub_cancel (u p) a (u q)).symm
     _ ≤ |u p - a|  + |a - u q| :=
           abs_add_le (u p - a) (a - u q)
     _ = |u p - a|  + |u q - a| :=
           congrArg (|u p - a| + ·) (abs_sub_comm a (u q))
     _ < ε / 2 + ε / 2 :=
           add_lt_add (hk p hp) (hk q hq)
     _ = ε :=
           add_halves ε

-- Lemmas used
-- ===========

variable (a b c d : ℝ)
variable (f : ℝ → ℝ)
#check (abs_add_le a b : |a + b| ≤ |a| + |b|)
#check (abs_sub_comm a b : |a - b| = |b - a|)
#check (add_halves a : a / 2 + a / 2 = a)
#check (add_left_cancel_iff : a + b = a + c ↔ b = c)
#check (add_lt_add : a < b → c < d → a + c < b + d)
#check (congrArg f : a = b → f a = f b)
#check (half_pos : 0 < a → 0 < a / 2)
#check (sub_add_sub_cancel a b c : (a - b) + (b - c) = a - c)
